// Sign-in / cloud state bootstrap. No confidential data is loaded until
// the signed-in user is verified against the private app_admins table.
async function establishUser(){
 if(recoveryMode){enterRecoveryMode();return;}
 if(!db){authMessage.textContent='Supabase library failed to load. Check internet connection and reload.';return;}
 const {data:{user},error:userError}=await db.auth.getUser();
 if(userError||!user){gate.classList.remove('hidden');authMessage.textContent='Please sign in with your authorised account.';return;}
 const {data:allowed,error:roleError}=await db.rpc('dashboard_is_admin');
 if(roleError||allowed!==true){await db.auth.signOut();gate.classList.remove('hidden');authMessage.textContent='This account does not have dashboard administrator access.';return;}
 try{
  cloudUser=user;
  await loadFromCloud();ready=true;gate.classList.add('hidden');render();
 }catch(err){gate.classList.remove('hidden');authMessage.textContent='Cloud setup incomplete: '+(err.message||err);}
}
document.getElementById('cloudLoginForm').addEventListener('submit',async event=>{
 event.preventDefault();if(!db)return;
 const button=document.getElementById('cloudLoginButton');button.disabled=true;
 authMessage.textContent='Signing in…';
 const email=document.getElementById('cloudEmail').value.trim();
 const password=document.getElementById('cloudPassword').value;
 try{const {error}=await db.auth.signInWithPassword({email,password});if(error)throw error;await establishUser();}
 catch(err){authMessage.textContent='Sign-in failed: '+(err.message||err)}
 finally{document.getElementById('cloudPassword').value='';button.disabled=false}
});
recoveryRequest.addEventListener('click',async()=>{
 if(!db){authMessage.textContent='Cannot connect to Supabase. Check internet connection.';return;}
 const email=document.getElementById('cloudEmail').value.trim();
 if(!email){authMessage.textContent='Enter your work email above to request a reset link.';return;}
 recoveryRequest.disabled=true;authMessage.textContent='Requesting password recovery email…';
 try{
  const redirectTo=window.location.origin+window.location.pathname;
  const {error}=await db.auth.resetPasswordForEmail(email,{redirectTo});
  if(error)throw error;
  authMessage.textContent='If the account is eligible, a password recovery link has been sent. Use the newest email.';
 }catch(err){authMessage.textContent='Could not request reset: '+(err.message||err);}
 finally{recoveryRequest.disabled=false;}
});
recoveryForm.addEventListener('submit',async event=>{
 event.preventDefault();if(!db)return;
 const pwd=document.getElementById('cloudNewPassword').value;
 const confirmPwd=document.getElementById('cloudConfirmPassword').value;
 if(pwd!==confirmPwd){authMessage.textContent='Passwords do not match.';return;}
 if(pwd.length<12){authMessage.textContent='Use at least 12 characters.';return;}
 const button=document.getElementById('cloudRecoveryButton');button.disabled=true;
 try{
  const {data:{session},error:sessionError}=await db.auth.getSession();
  if(sessionError||!session)throw new Error('Recovery link invalid or expired. Request a fresh link.');
  const {error}=await db.auth.updateUser({password:pwd});
  if(error)throw error;
  await db.auth.signOut();
  window.history.replaceState(null,'',window.location.pathname);
  recoveryMode=false;
  loginForm.hidden=false;recoveryRequest.hidden=false;recoveryForm.hidden=true;
  authMessage.textContent='Dashboard password updated. Sign in using the new password.';
 }catch(err){authMessage.textContent='Password update failed: '+(err.message||err);}
 finally{
  document.getElementById('cloudNewPassword').value='';
  document.getElementById('cloudConfirmPassword').value='';
  button.disabled=false;
 }
});
